\documentclass{book}

\usepackage{xassoccnt}

\usepackage{chngcntr}


\makeatletter

\AtBeginDocument{%
  \let\latex@@chapter\chapter
  \RenewDocumentCommand{\chapter}{sO{#3}m}{%
    \IfBooleanTF{#1}{%
      \latex@@chapter*{#3}%
    }{%
      \ifnogobblechapter
      \latex@@chapter[#2]{#3}
      \fi
     }%
   }
 }
\makeatother
 
\newcommand{\counterwithoutwrapper}[2]{%
  \counterwithout{#2}{#1}%
}

\newif\ifnogobblechapter
\nogobblechapterfalse

\usepackage{hyperref}

\begin{document}

\ifnogobblechapter
\else
\ClearCounterResetList*{chapter}

\fi
